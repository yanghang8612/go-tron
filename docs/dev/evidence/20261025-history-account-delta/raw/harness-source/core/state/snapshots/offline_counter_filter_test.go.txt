package snapshots

import (
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"reflect"
	"testing"
	"time"

	"github.com/tronprotocol/go-tron/common"
)

func TestOfflineCounterLogAddressFilter(t *testing.T) {
	dir := os.Getenv("GTRON_EVENT_COPY")
	if dir == "" {
		t.Skip("offline event copy required")
	}
	raw, err := os.ReadFile(filepath.Join(dir, "source-ref.json"))
	if err != nil {
		t.Fatal(err)
	}
	var ref SegmentRef
	if err := json.Unmarshal(raw, &ref); err != nil {
		t.Fatal(err)
	}
	if err := CheckEventLogSegment(dir, ref); err != nil {
		t.Fatal(err)
	}
	seg, err := OpenEventLogSegment(dir, ref)
	if err != nil {
		t.Fatal(err)
	}
	defer seg.Close()
	collect := func(f EventLogFilter) ([]EventLog, float64) {
		var rows []EventLog
		start := time.Now()
		err := seg.IterateLogs(3144021, 3144021, f, func(row EventLog) (bool, error) { rows = append(rows, row); return true, nil })
		if err != nil {
			t.Fatal(err)
		}
		return rows, float64(time.Since(start).Nanoseconds()) / 1e6
	}
	all, unfilteredMS := collect(EventLogFilter{})
	owner := common.BytesToAddress(common.FromHex("41a4997124a328a4615eab437cede31da47a31b375"))
	topic := common.BytesToHash(common.FromHex("a0d1e73d61e6eb8b3dafafe7c97e9471ec8e405c1ae705ac6aafaf4dd9abd6ae"))
	filter := EventLogFilter{Addresses: []common.Address{owner}, Topics: [][]common.Hash{{topic}}}
	var expected []EventLog
	for _, row := range all {
		if row.Address.AccountID() == owner.AccountID() && eventLogTopicsMatch(filter.Topics, row.Log.Topics) {
			expected = append(expected, row)
		}
	}
	if len(expected) != 1 || len(expected[0].Log.Address) != 20 {
		t.Fatalf("canonical witness: %+v", expected)
	}
	got, filteredMS := collect(filter)
	if !reflect.DeepEqual(expected, got) {
		t.Fatalf("filtered %d vs expected %d", len(got), len(expected))
	}
	index, err := BuildEventLogIndexSegmentFromEventLogSegments(dir, []SegmentRef{ref}, "")
	if err != nil {
		t.Fatal(err)
	}
	idx, err := OpenEventLogIndexSegment(dir, index)
	if err != nil {
		t.Fatal(err)
	}
	defer idx.Close()
	starts, used, err := idx.CandidateSegmentStarts(filter)
	if err != nil || !used || len(starts) != 1 || starts[0] != ref.FromTxNum {
		t.Fatalf("global index: %v/%v/%v", starts, used, err)
	}
	payload, _ := json.Marshal(got)
	out := map[string]any{"block": 3144021, "rows": len(got), "protobuf_address_bytes": len(got[0].Log.Address), "row_sha256": fmt.Sprintf("%x", sha256.Sum256(payload)), "original_payload_equal": true, "source_ref": ref, "global_index_candidates": starts, "unfiltered_ms": unfilteredMS, "filtered_ms": filteredMS}
	raw, _ = json.MarshalIndent(out, "", "  ")
	if err := os.WriteFile(os.Getenv("GTRON_EVENT_OUT"), raw, 0600); err != nil {
		t.Fatal(err)
	}
}
