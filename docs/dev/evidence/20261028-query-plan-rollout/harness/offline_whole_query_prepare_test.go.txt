package snapshots

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"reflect"
	"syscall"
	"testing"
	"time"

	"github.com/tronprotocol/go-tron/core/rawdb"
)

func TestOfflineWholeQueryPrepare(t *testing.T) {
	input, out := os.Getenv("GTRON_WHOLE_QUERY_INPUT"), os.Getenv("GTRON_WHOLE_QUERY_OUTPUT")
	if input == "" {
		t.Skip("private fixed copy only")
	}
	ctx := context.Background()
	save := func(name string, v any) {
		b, e := json.MarshalIndent(v, "", "  ")
		if e != nil {
			t.Fatal(e)
		}
		if e = os.WriteFile(filepath.Join(out, name), b, 0644); e != nil {
			t.Fatal(e)
		}
	}
	cpu := func() float64 {
		var r syscall.Rusage
		if e := syscall.Getrusage(syscall.RUSAGE_SELF, &r); e != nil {
			t.Fatal(e)
		}
		return float64(r.Utime.Sec+r.Stime.Sec) + float64(r.Utime.Usec+r.Stime.Usec)/1e6
	}
	dir := filepath.Join(input, "state-snapshots")
	original, e := LoadProductionManifest(dir)
	if e != nil {
		t.Fatal(e)
	}
	stable := func(m *Manifest) []byte {
		var refs []SegmentRef
		for _, r := range m.Segments {
			if r.Kind != SegmentAccountHistoryRoutingDelta {
				refs = append(refs, r)
			}
		}
		b, _ := json.Marshal(refs)
		return b
	}
	canonical := stable(original)
	coverage := func() (int, int) {
		m, e := LoadProductionManifest(dir)
		if e != nil {
			t.Fatal(e)
		}
		m = manifestLookupView(m)
		known, e := accountRoutingCoveredForBuild(ctx, dir, m)
		if e != nil {
			t.Fatal(e)
		}
		sources := manifestHistorySources(m)
		if sources == nil {
			t.Fatal("missing source table")
		}
		covered := 0
		for _, s := range sources.sources {
			if _, ok := known[s.id]; ok {
				covered++
			}
		}
		return covered, len(sources.sources)
	}
	before, total := coverage()
	state := accountRoutingRefreshState{}
	var passes []map[string]any
	start := time.Now()
	for i := 0; i < 512; i++ {
		began, used := time.Now(), cpu()
		built, e := refreshAccountHistoryDeltaWithBudget(ctx, dir, &state, time.Now(), 64, smallSystemRoutingWorkLimit{bytes: 128 << 20, duration: 2 * time.Second})
		if e != nil {
			t.Fatal(e)
		}
		current, e := LoadProductionManifest(dir)
		if e != nil {
			t.Fatal(e)
		}
		if !bytes.Equal(canonical, stable(current)) || !reflect.DeepEqual(original.Progress, current.Progress) {
			t.Fatal("canonical refs/frontiers changed")
		}
		pass := map[string]any{"pass": i, "built": built, "seconds": time.Since(began).Seconds(), "cpu_seconds": cpu() - used, "sources": state.last.sources, "source_reads": state.last.sourceReads, "input_bytes": state.last.inputBytes, "metadata_bytes": state.last.metadataBytes, "next_tx": state.nextTx, "unsupported": state.last.unsupported}
		passes = append(passes, pass)
		save("account-progress.json", map[string]any{"complete": false, "before": before, "total": total, "passes": passes})
		if !built {
			covered, n := coverage()
			if covered != n {
				t.Fatal("unrepaired account sources", covered, n, state)
			}
			break
		}
		if i == 511 {
			t.Fatal("bounded pass limit")
		}
	}
	merges := 0
	for ; merges < 256; merges++ {
		m, e := LoadProductionManifest(dir)
		if e != nil {
			t.Fatal(e)
		}
		n := 0
		for _, r := range m.Segments {
			if r.Kind == SegmentAccountHistoryRoutingDelta {
				n++
			}
		}
		if n <= 96 {
			break
		}
		ref, replace, built, e := compactAccountRoutingDeltas(ctx, dir, manifestLookupView(m), 128<<20)
		if e != nil || !built {
			t.Fatal("bounded delta consolidation", built, e)
		}
		if _, e = publishAccountHistoryDelta(ctx, dir, m, ref, replace); e != nil {
			t.Fatal(e)
		}
	}
	after, _ := coverage()
	save("account-result.json", map[string]any{"complete": true, "before": before, "after": after, "total": total, "passes": passes, "consolidation_passes": merges, "seconds": time.Since(start).Seconds(), "canonical_refs_unchanged": true, "frontiers_unchanged": true})
	tune := rawdb.DefaultPebbleOptions()
	tune.DisableAutomaticCompactions = true
	tune.MemTableSizeBytes = 32 << 20
	tune.MaxConcurrentCompactions = 1
	hot, e := rawdb.NewPebbleDBWithOptions(filepath.Join(input, "chaindata"), 128, 256, tune)
	if e != nil {
		t.Fatal(e)
	}
	defer hot.Close()
	tune.EnableL6Filter = false
	history, e := rawdb.NewPebbleDBWithOptions(filepath.Join(input, "history"), 128, 256, tune)
	if e != nil {
		t.Fatal(e)
	}
	defer history.Close()
	if e = rawdb.EnsureDirectHistoryPair(hot, history, false); e != nil {
		t.Fatal(e)
	}
	db := rawdb.NewDirectHistoryStore(hot, history)
	head, present, e := rawdb.ReadStateCommitmentHead(db)
	if e != nil || !present {
		t.Fatal(present, e)
	}
	indexed, present, e := rawdb.ReadStageProgress(db, rawdb.StageStateHistoryIndex)
	if e != nil || !present || indexed > head.BlockNum {
		t.Fatal(indexed, present, e)
	}
	view, release, e := rawdb.AcquireStateHistoryReadView(db)
	if e != nil {
		t.Fatal(e)
	}
	batch := db.NewBatch()
	built, skipped, size := 0, 0, 0
	began, used := time.Now(), cpu()
	for block := indexed + 1; block <= head.BlockNum; block++ {
		ok, e := rawdb.BuildStateHistoryOwnerSpans(ctx, view, batch, block)
		if e != nil {
			release()
			t.Fatal(block, e)
		}
		if ok {
			built++
		} else {
			skipped++
		}
		if batch.ValueSize() >= 8<<20 || block%100 == 0 {
			size += batch.ValueSize()
			if e = batch.Write(); e != nil {
				release()
				t.Fatal(e)
			}
			batch.Reset()
		}
		if block%100 == 0 {
			save("tail-progress.json", map[string]any{"complete": false, "indexed": indexed, "head": head.BlockNum, "block": block, "built": built, "tiny_or_missing": skipped, "value_bytes": size, "seconds": time.Since(began).Seconds()})
		}
	}
	size += batch.ValueSize()
	if e = batch.Write(); e != nil {
		release()
		t.Fatal(e)
	}
	release()
	for _, handle := range []any{hot, history} {
		f, ok := handle.(interface{ OfflineFlushKeyValue() error })
		if !ok {
			t.Fatal("missing diagnostic flush bridge")
		}
		if e = f.OfflineFlushKeyValue(); e != nil {
			t.Fatal(e)
		}
	}
	actual, present, e := rawdb.ReadStateCommitmentHead(db)
	if e != nil || !present || !reflect.DeepEqual(head, actual) {
		t.Fatal("commitment HEAD changed", e)
	}
	finalIndex, _, e := rawdb.ReadStageProgress(db, rawdb.StageStateHistoryIndex)
	if e != nil || finalIndex != indexed {
		t.Fatal("index frontier changed", e)
	}
	save("tail-result.json", map[string]any{"complete": true, "indexed": indexed, "head": head.BlockNum, "head_tx": head.TxNum, "root": fmt.Sprintf("%x", head.Root[:]), "built": built, "tiny_or_missing": skipped, "value_bytes": size, "seconds": time.Since(began).Seconds(), "cpu_seconds": cpu() - used, "commitment_head_unchanged": true, "frontier_unchanged": true, "private_memtables_flushed": true})
}
