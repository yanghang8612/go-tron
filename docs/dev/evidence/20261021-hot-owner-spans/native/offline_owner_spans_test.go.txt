const ownerTailFrom uint64 = 11817531
const ownerTailThrough uint64 = 11818192

type ownerSpansOnlyBatch struct { batch ethdb.Batch; rows, size uint64 }
func (w *ownerSpansOnlyBatch) Delete(k []byte) error {return errors.New("deletes prohibited in derived builder")}
func (w *ownerSpansOnlyBatch) Put(k,v []byte) error {
 if len(k)!=len(stateChangeOwnerSpansPrefix)+8 || !bytes.HasPrefix(k,stateChangeOwnerSpansPrefix) {return errors.New("non-auxiliary write prohibited")}
 block:=binary.BigEndian.Uint64(k[len(stateChangeOwnerSpansPrefix):]);if block<ownerTailFrom || block>ownerTailThrough{return errors.New("write outside fixed hot tail prohibited")}
 w.rows++;w.size+=uint64(len(k)+len(v));return w.batch.Put(k,v)
}
func ownerWriteDigest(h hash.Hash,k,v []byte) { var n [8]byte;binary.BigEndian.PutUint64(n[:],uint64(len(k)));h.Write(n[:]);h.Write(k);binary.BigEndian.PutUint64(n[:],uint64(len(v)));h.Write(n[:]);h.Write(v) }
func ownerCanonicalDigest(db StateHistoryReadView) (map[string]any,error) {
 result:=map[string]any{}
 for _,family:=range DirectHistoryRanges() {
  if family.Name!="state_changeset" && family.Name!="state_tx_range" && family.Name!="state_history_chunk" && family.Name!="state_history_chunk_bucket" {continue}
  first,last:=ownerTailFrom,ownerTailThrough
  if family.Name=="state_history_chunk" || family.Name=="state_history_chunk_bucket" {first/=StateHistoryChunkBucketBlocks;last/=StateHistoryChunkBucketBlocks}
  var start [8]byte;binary.BigEndian.PutUint64(start[:],first)
  h:=sha256.New();var rows,size uint64;it:=db.NewIterator(family.Start,start[:])
  for it.Next() {k,v:=it.Key(),it.Value();if len(k)<len(family.Start)+8{return nil,errors.New("short canonical key")};ordinal:=binary.BigEndian.Uint64(k[len(family.Start):]);if ordinal>last{break};ownerWriteDigest(h,k,v);rows++;size+=uint64(len(k)+len(v))};e:=it.Error();it.Release();if e!=nil{return nil,e};result[family.Name]=map[string]any{"rows":rows,"logical_bytes":size,"sha256":hex.EncodeToString(h.Sum(nil)),"from":first,"through":last}
 }
 return result,nil
}
